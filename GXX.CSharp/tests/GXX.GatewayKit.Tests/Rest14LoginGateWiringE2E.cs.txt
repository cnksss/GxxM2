``csharpusing System;
using System.IO;
using GXX.GatewayKit.Rest11;
using Xunit;

namespace GXX.GatewayKit.Tests;

/// <summary>
/// 杞﹂亾 **p14-logingate-wire** 鐨?*绔埌绔帴绾?*鐢ㄤ緥锛氱洿鎺ユ瀯閫犵湡瀹炵殑
/// `GXX.LoginGate.LoginGateService`锛岄獙璇?榛樿鍏抽棴"涓?鏄惧紡 opt-in"涓や晶銆?/// 锛堟湰鏂囦欢鍦?csproj 澧炲姞 `GXX.LoginGate` 寮曠敤鏃舵墠鍙備笌缂栬瘧锛涜鎶ュ憡 搂闃诲銆傦級
/// </summary>
public class ZWireTests : IDisposable
{
    private readonly string _dir;

    public ZWireTests()
    {
        _dir = Path.Combine(Path.GetTempPath(), "zwire-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(_dir);
    }

    public void Dispose()
    {
        try { Directory.Delete(_dir, true); } catch { }
    }

    private string IniPath => Path.Combine(_dir, "Config.ini");

    [Fact]
    public void DefaultConstructor_DoesNotWireRest11()
    {
        var svc = new GXX.LoginGate.LoginGateService(null, IniPath);
        try
        {
            Assert.Null(svc.Rest11Options);
            Assert.Null(svc.Rest11Wire);
        }
        finally { svc.Dispose(); }
    }

    [Fact]
    public void OptInConstructor_WiresKernelAndLoadsOriginalSections()
    {
        File.WriteAllText(IniPath,
            "[LoginGate]\r\nShowLogLevel=6\r\nCount=1\r\nServerAddr=127.0.0.1\r\nServerPort=5500\r\nGatePort=7000\r\n" +
            "[Integer]\r\nMaxConnectOfIP=3\r\nCheckNewIDOfIP=2\r\nClientTimeOutTime3=20000\r\n" +
            "[Switch]\r\nCheckNullSession=-1\r\nCheckNewIDOfIP=-1\r\n\r\n");

        var svc = new GXX.LoginGate.LoginGateService(Rest11LoginGateOptions.All, IniPath);
        try
        {
            Assert.NotNull(svc.Rest11Options);
            Assert.NotNull(svc.Rest11Wire);
            Assert.True(svc.Rest11Wire!.Enabled);

            Assert.Equal(6, svc.Rest11Wire.g_pConfig.m_nShowLogLevel);
            Assert.Equal(3, svc.Rest11Wire.g_pConfig.m_nMaxConnectOfIP);
            Assert.Equal(2, svc.Rest11Wire.g_pConfig.m_nCheckNewIDOfIP);
            Assert.Equal(20000, svc.Rest11Wire.g_pConfig.m_nClientTimeOutTime);

            // `GateService.IRest11GateEnforcement` 鎺ョ紳鐨勪笁鏉″垽瀹氬湪鐪熷疄 kernel 涓婂彲璋冪敤
            Assert.False(svc.Rest11Wire.IsBlockIP("10.0.0.1"));
            Assert.False(svc.Rest11Wire.IsBlockIPArea("10.0.0.1"));
            Assert.False(svc.Rest11Wire.OverConnectOfIP("10.0.0.1"));
        }
        finally { svc.Dispose(); }
    }

    [Fact]
    public void OptIn_StartService_RunsResidualStartup()
    {
        File.WriteAllText(IniPath,
            "[LoginGate]\r\nShowLogLevel=6\r\nCount=1\r\nGateAddr=127.0.0.1\r\nGatePort=0\r\nServerPort=5500\r\n" +
            "[Switch]\r\nCheckNullSession=-1\r\n\r\n");

        var svc = new GXX.LoginGate.LoginGateService(Rest11LoginGateOptions.All, IniPath);
        try
        {
            Assert.True(svc.StartService());
            Assert.True(svc.Rest11Wire!.g_fServiceStarted);
            Assert.True(svc.Rest11Wire.g_ProcMsgThread.Enabled);
            Assert.True(File.Exists(svc.Rest11Wire.IPAddrFilter.BlockFilePath));
        }
        finally
        {
            svc.StopService();
            svc.Dispose();
        }
    }
}

``
